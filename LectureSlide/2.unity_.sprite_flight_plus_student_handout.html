2. 최고점수를 저장하는 메서드 작성

using UnityEngine;

public class GameManager : MonoBehaviour
{
    public void SaveHighScore(int score)
    {
        int currentHighScore =
            PlayerPrefs.GetInt("HighScore", 0);

        if (score > currentHighScore)
        {
            PlayerPrefs.SetInt("HighScore", score);
            PlayerPrefs.Save();

            Debug.Log(
                "New high score saved: " + score
            );
        }
    }
}
핵심 개념
"HighScore"는 데이터를 찾기 위한 Key이고, 0은 아직 저장된 값이 없을 때 사용할 기본값입니다.

public class GameManager : MonoBehaviour

5. 점수에 따라 Difficulty Level 계산

비어 있는 기존 Update()를 다음과 같이 수정하세요.

void Update()
{
    if (player == null || rb == null)
        return;

    int score = player.CurrentScore;

    int newLevel = 1;

    if (score >= 150)
    {
        newLevel = 4;
    }
    else if (score >= 100)
    {
        newLevel = 3;
    }
    else if (score >= 50)
    {
        newLevel = 2;
    }

    if (newLevel != currentDifficultyLevel)
    {
        currentDifficultyLevel = newLevel;
        ApplyDifficulty();
    }
}

6. Level별 속도 배율 적용

void ApplyDifficulty()
{
    float speedMultiplier = 1f;

    switch (currentDifficultyLevel)
    {
        case 2:
            speedMultiplier = 1.25f;
            break;

        case 3:
            speedMultiplier = 1.5f;
            break;

        case 4:
            speedMultiplier = 1.8f;
            break;
    }

    if (rb.linearVelocity.magnitude > 0.01f)
    {
        rb.linearVelocity =
            rb.linearVelocity.normalized *
            baseVelocity *
            speedMultiplier;
    }

    Debug.Log(
        "Difficulty Level: " +
        currentDifficultyLevel
    );
}
정상 결과
49점까지는 기존 속도 → 50점에서 한 번 빨라짐 → 100점에서 다시 빨라짐 → 150점부터 최고 난이도로 변경됩니다. Console에서도 Level 변경을 확인할 수 있습니다.

실습에 더 안정적인 Obstacle.cs 구현

기존 AddForce() 방식 대신 시작 속도를 직접 지정하면 baseVelocity를 정확히 기억할 수 있어 실습 중 결과가 일정합니다. Start()의 이동 초기화 부분을 아래처럼 바꿔도 좋습니다. 

float randomSpeed =
    Random.Range(minSpeed, maxSpeed) / randomSize;

Vector2 randomDirection =
    Random.insideUnitCircle.normalized;

rb = GetComponent<Rigidbody2D>();

baseVelocity = randomSpeed;

rb.linearVelocity =
    randomDirection * baseVelocity;
핵심 개념
normalized는 방향만 유지하고 길이를 1로 만든 뒤, 우리가 원하는 속도 값을 다시 곱하는 방식입니다. 그래서 “방향”과 “속도”를 따로 생각할 수 있습니다.

Challenge A — 점수에 비례해서 계속 빨라지기

단계가 아니라 점수가 올라갈 때마다 조금씩 증가시키고 싶다면 Level 계산 대신 다음처럼 배율을 계산할 수 있습니다.

float speedMultiplier =

Challenge B — 최대 난이도 제한

계속 빨라지면 결국 플레이가 불가능해지므로 최대 배율을 제한합니다.

float speedMultiplier =
    1f + player.CurrentScore * 0.005f;

speedMultiplier =
    Mathf.Min(speedMultiplier, 2f);

또는 한 줄로:

float speedMultiplier =
    Mathf.Clamp(
        1f + player.CurrentScore * 0.005f,
        1f,
        2f
    );
주의
매 Frame마다 현재 속도에 다시 * 1.25를 하는 식으로 구현하면 속도가 기하급수적으로 증가합니다. 반드시 기준 속도 × 난이도 배율 형태로 계산해야 합니다.

5. OnCollisionEnter2D를 교체

void OnCollisionEnter2D(Collision2D collision)
{
    // 같은 장애물에 붙어서 HP가 한 번에 0이 되는 것을 방지
    if (Time.time - lastHitTime < damageCooldown)
        return;

    lastHitTime = Time.time;

    hp--;
    UpdateHpUI();

    if (hp <= 0)
    {
        Instantiate(
            explosionEffect,
            transform.position,
            transform.rotation
        );

        if (borderParent != null)
            borderParent.SetActive(false);

        restartButton.style.display = DisplayStyle.Flex;
        gameManager.SaveHighScore(Mathf.FloorToInt(score));

        Destroy(gameObject);
    }
}
중요
기존 OnCollisionEnter2D()에 있던 즉시 Destroy(gameObject)를 그대로 두면 HP를 만들었어도 첫 충돌에 죽습니다. 반드시 위 로직으로 교체합니다.
테스트
HP 3 → 2 → 1 → 0 순서로 줄고 0이 되었을 때만 Restart 버튼과 Game Over가 발생해야 합니다.

2. HeartSpawner.cs 생성

Assets/Scrips/HeartSpawner.cs
using System.Collections;
using UnityEngine;

public class HeartSpawner : MonoBehaviour
{
    public GameObject heartPrefab;

    public float minSpawnTime = 4f;
    public float maxSpawnTime = 8f;

    public float minX = -7f;
    public float maxX = 7f;
    public float minY = -4f;
    public float maxY = 4f;

    void Start()
    {
        StartCoroutine(SpawnLoop());
    }

    IEnumerator SpawnLoop()
    {
        while (true)
        {
            float waitTime =
                Random.Range(minSpawnTime, maxSpawnTime);

            yield return new WaitForSeconds(waitTime);

            Vector3 position = new Vector3(
                Random.Range(minX, maxX),
                Random.Range(minY, maxY),
                0f
            );

            Instantiate(
                heartPrefab,
                position,
                Quaternion.identity
            );
        }
    }
}

2. Heart.cs 생성

Assets/Scrips/Heart.cs
using UnityEngine;

public class Heart : MonoBehaviour
{
    private void OnTriggerEnter2D(Collider2D other)
    {
        PlayerContoller player =
            other.GetComponent<PlayerContoller>();

        if (player == null)
            return;

        player.CollectHeart();
        Destroy(gameObject);
    }
}
